-- Filtra particularidades de severidade CRÍTICA ou ATENÇÃO do pet selecionado
-- Exibidas em container de destaque no formulário de reserva
SELECT
    p2.*,
    p1.par001_desc,
    p2.par002_atencao AS severidade
FROM "{{JSutils.currentSchema()}}".hotpar002 p2
INNER JOIN "{{JSutils.currentSchema()}}".hotpar001 p1
    ON p1.par001_id = p2.par002_par001_id
WHERE p2.par002_deleted <> true
  AND p2.par002_ani002_id = {{ appsmith.store.petID_Reserva }}
  AND p2.par002_atencao = true
ORDER BY p1.par001_desc ASC;
