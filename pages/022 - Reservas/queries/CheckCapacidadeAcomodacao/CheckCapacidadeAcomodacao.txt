-- Retorna ocupação atual e capacidade máxima da acomodação no período.
-- Exclui o próprio registro em edição e reservas com situacao 3=checkout ou 4=cancelada.
-- ATENÇÃO: ajustar o nome da coluna de capacidade se necessário
--          (acomo001_capacidade ou acomo001_capaciade).
SELECT
    COUNT(*) AS ocupacao_atual,
    (
        SELECT acomo001_capacidade
        FROM "{{JSutils.currentSchema()}}".hotacomo001
        WHERE acomo001_id = {{appsmith.store.acomodacaoID_Reserva}}
    ) AS capacidade_maxima
FROM "{{JSutils.currentSchema()}}".hotreserva001
WHERE reserva001_deleted         <> true
  AND reserva001_situacao         NOT IN (3, 4)
  AND reserva001_acomo001_id      = {{appsmith.store.acomodacaoID_Reserva}}
  AND reserva001_id              != {{appsmith.store.idSelecionadoRESERVA || 0}}
  AND reserva001_prevdatain      <= '{{appsmith.store.checkoutReserva}}'
  AND COALESCE(
        reserva001_checkout,
        reserva001_prevdatain + INTERVAL '30 days'
      )                          >= '{{appsmith.store.checkinReserva}}';
