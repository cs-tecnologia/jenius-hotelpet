SELECT count(*) AS total
FROM "{{JSutils.currentSchema()}}".hotreserva001 r
WHERE r.reserva001_deleted <> true
AND (
  r.reserva001_checkout = CURRENT_DATE
  OR (
    r.reserva001_checkin IS NOT NULL
    AND r.reserva001_checkout IS NULL
    AND (r.reserva001_prevdatain + r.reserva001_prevdiashospedagem * INTERVAL '1 day')::date = CURRENT_DATE
  )
);
